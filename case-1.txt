# University Scholarship Allocation

students = []
eligible_students = []
ineligible_students = []

n = int(input("Enter number of students: "))

for i in range(n):
    print("\nEnter details for Student", i + 1)

    name = input("Student name: ")
    aggregate = float(input("Aggregate marks (%): "))

    subject1 = float(input("Subject 1 marks (%): "))
    subject2 = float(input("Subject 2 marks (%): "))
    subject3 = float(input("Subject 3 marks (%): "))

    attendance = float(input("Attendance (%): "))
    income = float(input("Annual family income (Rs.): "))

    student = {
        "name": name,
        "aggregate": aggregate,
        "attendance": attendance,
        "income": income
    }

    if (aggregate >= 75 and
        subject1 >= 60 and
        subject2 >= 60 and
        subject3 >= 60 and
        attendance >= 75 and
        income <= 250000):

        eligible_students.append(student)
        print("Status: Eligible")

    else:
        ineligible_students.append(student)
        print("Status: Not Eligible")
